/** Temporary probes of the 2026-10-29 transaction-path re-audit (deleted after the run). */
@OptIn(ExperimentalCoroutinesApi::class)
class ReauditProbeTest {
    @get:Rule val tmp = TemporaryFolder()
    private val network = NetworkInfo("testnet-v6", "Testnet", "6.0.0", "6.0.0", "node", "v6", true)
    private val pxe = FakePxe().apply { deployed = true }
    private val events = MutableSharedFlow<PxeEvent>(extraBufferCapacity = 16)
    private val scopes = mutableListOf<CoroutineScope>()
    private val token = TokenInfo("0xt", "Test", "TST", 6)
    private val recipient = "0x" + "cd".repeat(32)

    @After fun cleanup() {
        scopes.forEach { it.cancel() }
        ProofGate.reset()
    }

    private fun TestScope.session(onStop: () -> Unit = {}): WalletSession {
        val vault = Vault(tmp.root.resolve("vault"), PlainBox)
        val contents = if (vault.exists()) vault.unlock() else vault.create(AccountSecrets("main", "0x" + "ab".repeat(32), Vault.ZERO_SALT))
        val launcher = PxeLauncher { object : RunningPxe {
            override val client = PxeClient(pxe)
            override val events = this@ReauditProbeTest.events
            override fun stop() = onStop()
        } }
        val scope = CoroutineScope(StandardTestDispatcher(testScheduler) + SupervisorJob()).also { scopes += it }
        return WalletSession(FakeBackend, launcher, unlocked(vault, contents), tmp.root, network, scope)
    }

    /**
     * Probe 1: the hash is told (txSubmitting, with the row's id) and THEN the channel times out while node.sendTx hangs.
     * Expected: the row keeps its hash and is followed (PENDING), as the generic path does for any other error.
     */
    @Test fun aHashToldBeforeTheTimeoutIsKeptByTheUncertainPath() = runTest {
        val s = session().apply { open() }
        val stops = CompletableDeferred<Unit>()
        pxe.whileSending = {
            val id = s.store.proving().single().id
            events.emit(PxeEvent("txSubmitting", Json.parseToJsonElement("""{"from":"addr-abab","txHash":"0xtold","id":"$id"}""")))
            stops.await()
            throw PxeException("transfer: no answer from the wallet engine after 20 min", unanswered = true)
        }
        val sending = async { runCatching { s.transactions.send(TransferKind.PRIVATE, token, recipient, BigInteger.ONE) } }
        runCurrent()
        assertEquals("the hash was recorded before the timeout", "0xtold", s.store.proving().single().txHash)
        stops.complete(Unit)
        val error = sending.await().exceptionOrNull()
        val row = s.state.value.activity.single()
        println("PROBE1 error=${error?.javaClass?.simpleName} status=${row.status} txHash=${row.txHash} rowError=${row.error?.take(40)}")
        assertEquals("the hash told before the timeout must not be lost", "0xtold", row.txHash)
        assertEquals(ActivityStatus.PENDING, row.status)
    }

    /** Probe 1b: after a restart, such a row is shown Failed and is not followed. */
    @Test fun aHashToldBeforeTheTimeoutAfterARestart() = runTest {
        val s = session().apply { open() }
        val stops = CompletableDeferred<Unit>()
        pxe.whileSending = {
            val id = s.store.proving().single().id
            events.emit(PxeEvent("txSubmitting", Json.parseToJsonElement("""{"from":"addr-abab","txHash":"0xtold","id":"$id"}""")))
            stops.await()
            throw PxeException("transfer: no answer from the wallet engine after 20 min", unanswered = true)
        }
        val sending = async { runCatching { s.transactions.send(TransferKind.PRIVATE, token, recipient, BigInteger.ONE) } }
        runCurrent()
        stops.complete(Unit)
        sending.await()
        val id = s.state.value.activity.single().id
        s.close()
        val restarted = WalletStore(tmp.root, network.id)
        restarted.failInterrupted()
        println("PROBE1b status=${restarted.item(id)?.status} txHash=${restarted.item(id)?.txHash} error=${restarted.item(id)?.error?.take(40)} unsettled=${restarted.unsettled().size}")
        assertEquals("0xtold", restarted.item(id)?.txHash)
        assertEquals(ActivityStatus.PENDING, restarted.item(id)?.status)
    }

    /**
     * Probe 2 (ProofGate interleaving c): the retired host's late proof lands while the NEW host's wallet send is being
     * proven: it is refused and does not mark the new send as sending.
     */
    @Test fun aRetiredHostsLateProofDuringTheNextHostsSend() = runTest {
        var stopped = 0
        val s1 = session(onStop = { stopped++ }).apply { open() }
        pxe.whileProving = { throw PxeException("transfer: no answer from the wallet engine after 20 min", unanswered = true) }
        runCatching { s1.transactions.send(TransferKind.PRIVATE, token, recipient, BigInteger.ONE) }
        assertEquals(Phase.FAILED, s1.state.value.phase)
        assertEquals(1, stopped)
        assertFalse(ProofGate.cancelled); assertFalse(ProofGate.released.value)
        // Retry: a new session and host.
        s1.close()
        val s2 = session(onStop = { stopped++ }).apply { open() }
        var refused = false
        pxe.whileProving = {
            // The old host's proof finishes now: its handler calls release(hostAlive = false).
            refused = runCatching { ProofGate.release(hostAlive = false) }.exceptionOrNull() is CancelledByUser
            assertFalse("the old proof must not mark the new send as sending", ProofGate.released.value)
        }
        val sent = s2.transactions.send(TransferKind.PRIVATE, token, recipient, BigInteger.TWO)
        assertTrue(refused)
        assertEquals(ActivityStatus.PENDING, sent.status)
        assertEquals(1, pxe.sent.size)
        // After a dApp approval on the new host, the gate is open for its proof.
        ProofGate.reset(); ProofGate.check(); ProofGate.release(hostAlive = true)
        // Two retirements in a row: the second host is retired too, and the gate is open again after it.
        pxe.whileProving = { throw PxeException("transfer: no answer from the wallet engine after 20 min", unanswered = true) }
        runCatching { s2.transactions.send(TransferKind.PRIVATE, token, recipient, BigInteger.ONE) }
        assertEquals(Phase.FAILED, s2.state.value.phase)
        assertEquals(2, stopped)
        assertFalse(ProofGate.cancelled)
        assertThrows(CancelledByUser::class.java) { ProofGate.release(hostAlive = false) }
        ProofGate.check(); ProofGate.release(hostAlive = true)
    }

    /** Probe 3: while a wallet send is uncertain, a dApp proof still goes through and the auto-lock does not wait for it. */
    @Test fun anUncertainSendDoesNotBlockDappProofsNorTheAutoLock() = runTest {
        val s = session().apply { open() }
        pxe.whileSending = { throw PxeException("transfer: no answer", unanswered = true) }
        runCatching { s.transactions.send(TransferKind.PRIVATE, token, recipient, BigInteger.ONE) }
        assertEquals(Transactions.UNCERTAIN, s.store.proving().single().error)
        ProofGate.check(); ProofGate.release(hostAlive = true)
        assertFalse("the auto-lock does not wait for an uncertain wallet row", s.isSending)
        assertNull(s.transactions.proving.value)
    }
}
